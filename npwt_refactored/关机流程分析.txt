【核心机制】
长按确认键（KEY_CL / KEY_OKL）触发关机，保存配置到Flash，然后断电

关机模式处理：MOD_OFF
───────────────────────

case MOD_OFF:
{
    if ((bat_lev & 0x0f) == LOWER_THAN_3_5V)
    {
        // 电池电量过低，直接关机，不保存
        ;
    }
    else
    {
        if (close_tim++ > 50)  // 等待1秒（50×20ms）
        {
            // 1. 关闭背光和音频
            ClrBlk();              // 关闭背光
            AUDIO_EN = 0;          // 关闭音频使能
            
            // 2. 保存配置到Flash
            Flash_Erase(addr);     // 擦除Flash扇区
            
            // 保存各项配置：
            Write_One_Word(addr,    mod_seta_preh_bak);  // 高压目标值
            Write_One_Word(addr+2,  mod_seta_ont);       // 间歇高压时间
            Write_One_Word(addr+4,  mod_seta_oft);       // 间歇低压时间
            Write_One_Word(addr+6,  mod_main_baka);      // 工作模式
            
            // 保存压力校准系数K（浮点数转整数）
            dataK  = (unsigned int)(valueK * 1000.0);
            dataK1 = dataK / 1000;            // 个位
            dataK2 = dataK % 1000 / 100;      // 十分位
            dataK3 = (dataK % 1000) % 100 / 10;  // 百分位
            dataK4 = (dataK % 1000) % 100 % 10;  // 千分位
            Write_One_Word(addr+8,  dataK1);
            Write_One_Word(addr+10, dataK2);
            Write_One_Word(addr+12, dataK3);
            Write_One_Word(addr+14, dataK4);
            
            // 保存静音标志和语言
            key_silent_flag = (key_silent_flag | language);
            Write_One_Word(addr+16, key_silent_flag);
            
            // 保存低压目标值
            Write_One_Word(addr+18, mod_seta_prel);
            
            // 3. 断电
            POWER_ON = 0;  // 关键：释放电源自锁，设备断电
        }
    }
}

================================================================================
关键参数和宏定义
================================================================================

按键定义：
  KEY_C     = 0x38  // 确认键短按
  KEY_CL    = 0xb8  // 确认键长按（0x38 + 0x80）
  KEY_OKL   = 0xb8  // 同 KEY_CL

长按检测：
  JUDGE_LONG_PRESS_TIME = 100  // 100个周期 = 2秒

Flash地址：
  addr = 0xa000  // Flash起始地址
  
保存的配置：
  addr+0:  高压目标值（mod_seta_preh_bak）
  addr+2:  间歇高压时间（mod_seta_ont）
  addr+4:  间歇低压时间（mod_seta_oft）
  addr+6:  工作模式（mod_main_baka）
  addr+8:  压力校准系数K1（个位）
  addr+10: 压力校准系数K2（十分位）
  addr+12: 压力校准系数K3（百分位）
  addr+14: 压力校准系数K4（千分位）
  addr+16: 静音标志和语言选择
  addr+18: 低压目标值（mod_seta_prel）

电源控制：
  POWER_ON = LATCbits.LATC2  // RC2引脚
  POWER_ON = 0 → 断电
  POWER_ON = 1 → 保持电源


================================================================================
关机时间线
================================================================================

t=0ms: 检测到长按确认键（KEY_CL = 0xb8）
  ↓
  进入 MOD_OFF 模式
  清屏显示
  ↓
t=0~1000ms: 等待1秒（close_tim 计数到50）
  ↓
t=1000ms: 执行关机动作
  1. 关闭背光（ClrBlk()）
  2. 关闭音频（AUDIO_EN = 0）
  3. 擦除Flash扇区（Flash_Erase）
  4. 保存所有配置参数到Flash
  5. 关闭LCD显示（JLX12864G_OFF）
  6. 断电（POWER_ON = 0）


================================================================================
特殊情况：低电量关机
================================================================================

如果电池电量 < 3.5V：
  - 不保存配置（避免Flash写入失败）
  - 显示低电量警告
  - 直接进入 MOD_OFF 状态
  - 等待自动断电或手动关机


================================================================================
锁定功能与关机
================================================================================

解锁按键：
  UNLOCK_NPWT_KEYVAL   = 0x24  // 解锁短按
  UNLOCK_NPWT_L_KEYVAL = 0xa4  // 解锁长按（0x24 + 0x80）

锁定状态（lock_flg = 1）时：
  - 按键被锁定，无法关机
  - 需要先解锁（按解锁键）
  - 解锁后才能长按确认键关机


================================================================================
重构工程需要实现的逻辑
================================================================================

【核心要求】
1. 检测长按确认键（≥2秒）
2. 进入关机模式（MODE_SHUTDOWN）
3. 等待1秒
4. 保存所有配置到Flash：
   - 压力设置（高压、低压）
   - 间歇模式时间（高压时间、低压时间）
   - 工作模式
   - 压力校准系数
   - 静音标志和语言
5. 关闭所有外设：
   - 气泵、电磁阀
   - 背光、LCD
   - 音频、蜂鸣器
6. 断电（POWER_ON = 0）

void State_Shutdown_Handler(SystemState_t *state)
{
    static uint16_t shutdown_timer = 0;
    static bool flash_saved = false;
    
    shutdown_timer++;
    
    /* 阶段1：立即关闭外设 */
    if (shutdown_timer == 1)
    {
        /* 停止所有工作 */
        HAL_Pump_Stop();
        HAL_Valve1_Close();
        HAL_Valve2_Close();
        HAL_Buzzer_Off();
        
        /* 关闭背光 */
        LCD_SetBacklight(false);
        
        /* 清屏 */
        LCD_Clear();
    }
    
    /* 阶段2：等待1秒后保存配置 */
    if (shutdown_timer >= 50 && !flash_saved)  // 1秒后
    {
        /* 保存配置到Flash */
        Flash_SaveSystemSettings(state);
        flash_saved = true;
        
        /* 关闭LCD */
        /* LCD_PowerOff(); */
    }
    
    /* 阶段3：断电 */
    if (shutdown_timer >= 60)  // 1.2秒后
    {
        /* 释放电源自锁 */
        HAL_Power_Release();  // POWER_ON = 0
        
        /* 复位计时器（实际不会执行到，因为已断电） */
        shutdown_timer = 0;
        flash_saved = false;
    }
}
